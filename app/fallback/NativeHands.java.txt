package com.pocketvr.app;

import android.content.Context;

/** Stand-in used only when the real native hand model could not be built: the page then keeps using its own detection. */
public class NativeHands {
    public interface Cb { void done(boolean ok, Object data); }
    public void init(Context ctx, String delegate, int hands, Cb cb) { cb.done(false, "ไม่มีตัวตรวจเนทีฟในแอปนี้"); }
    public void detect(String jpgBase64, long ts, String mode, Cb cb) { cb.done(false, "ไม่มีตัวตรวจเนทีฟในแอปนี้"); }
    public void close() { }
}
